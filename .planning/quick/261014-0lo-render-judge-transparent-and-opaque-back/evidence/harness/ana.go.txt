package main

import (
	"fmt"
	"image"
	"image/color"
	"image/png"
	"os"
)

func load(p string) image.Image {
	f, err := os.Open(p)
	if err != nil {
		panic(err)
	}
	defer f.Close()
	img, err := png.Decode(f)
	if err != nil {
		panic(err)
	}
	return img
}

func at(img image.Image, x, y int) color.NRGBA {
	return color.NRGBAModel.Convert(img.At(x, y)).(color.NRGBA)
}

func comp(c color.NRGBA) (int, int, int) {
	a := int(c.A)
	f := func(v uint8) int { return (int(v)*a + 255*(255-a) + 127) / 255 }
	return f(c.R), f(c.G), f(c.B)
}

func main() {
	o, t := load(os.Args[1]), load(os.Args[2])
	b := o.Bounds()
	var alpha0, alpha255, alphaPart, alpha0NonWhite int
	var oPart, tAlpha, tComp, both, onlyO, onlyT, oDark, tDark int
	var maxDiff, diffPx int
	x0, y0, x1, y1 := 1<<30, 1<<30, -1, -1
	tx0, ty0, tx1, ty1 := 1<<30, 1<<30, -1, -1
	var bright255 int
	for y := 0; y < b.Dy(); y++ {
		for x := 0; x < b.Dx(); x++ {
			oc, tc := at(o, x, y), at(t, x, y)
			switch tc.A {
			case 0:
				alpha0++
				if tc.R != 255 || tc.G != 255 || tc.B != 255 {
					alpha0NonWhite++
				}
			case 255:
				alpha255++
			default:
				alphaPart++
			}
			op := oc.R < 250 || oc.G < 250 || oc.B < 250
			r, g, bl := comp(tc)
			tp := r < 250 || g < 250 || bl < 250
			if tc.A > 0 {
				tAlpha++
				if !tp {
					bright255++
				}
			}
			if op {
				oPart++
				if x < x0 { x0 = x }
				if x > x1 { x1 = x }
				if y < y0 { y0 = y }
				if y > y1 { y1 = y }
			}
			if tp {
				tComp++
				if x < tx0 { tx0 = x }
				if x > tx1 { tx1 = x }
				if y < ty0 { ty0 = y }
				if y > ty1 { ty1 = y }
			}
			if oc.R < 80 && oc.G < 80 && oc.B < 80 { oDark++ }
			if r < 80 && g < 80 && bl < 80 { tDark++ }
			switch {
			case op && tp:
				both++
			case op:
				onlyO++
			case tp:
				onlyT++
			}
			d := max(abs(int(oc.R)-r), abs(int(oc.G)-g), abs(int(oc.B)-bl))
			if d > 0 {
				diffPx++
			}
			maxDiff = max(maxDiff, d)
		}
	}
	union := both + onlyO + onlyT
	fmt.Printf("alpha0=%d alpha0_nonwhite_rgb=%d alpha255=%d alpha_partial=%d t_alpha_gt0=%d t_alpha_gt0_but_composite_white=%d\n", alpha0, alpha0NonWhite, alpha255, alphaPart, tAlpha, bright255)
	fmt.Printf("opaque_part=%d transparent_composite_part=%d both=%d only_opaque=%d only_transparent=%d symdiff_over_union=%.4f\n", oPart, tComp, both, onlyO, onlyT, float64(onlyO+onlyT)/float64(max(union, 1)))
	fmt.Printf("opaque_box=%d,%d..%d,%d transparent_box=%d,%d..%d,%d dark_o=%d dark_t=%d\n", x0, y0, x1, y1, tx0, ty0, tx1, ty1, oDark, tDark)
	fmt.Printf("pixels_differing_opaque_vs_composite=%d max_channel_diff=%d\n", diffPx, maxDiff)
}

func abs(v int) int {
	if v < 0 {
		return -v
	}
	return v
}
