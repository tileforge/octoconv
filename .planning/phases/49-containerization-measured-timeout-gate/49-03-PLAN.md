  PANDOC_ENGINE_TIMEOUT = ceil(worst_cell_p95 x 2.0)   // 2.0 safety, mirrors ocr/av; NO MAX_PAGES multiplier (pandoc cost is per-conversion, not per-page)
  CONSTRAINT: PANDOC_ENGINE_TIMEOUT < 900  (RECONCILER_ACTIVE_STALE_AFTER = 15m)

  stop_grace_period = PANDOC_ENGINE_TIMEOUT + 10s (ShutdownTimeout margin) + 10s (Docker SIGKILL margin)
